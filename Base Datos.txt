);

CREATE TABLE Categoria (
    IdCategoria INT PRIMARY KEY, NombreCategoria VARCHAR(100), TipoCategoria VARCHAR(50)
);

CREATE TABLE MedioPago (

);

    IdTransaccion INT AUTO_INCREMENT PRIMARY KEY,

    MontoIngreso DECIMAL(10,2), MontoGasto DECIMAL(10,2), MontoAhorro DECIMAL(10,2), MontoDeuda DECIMAL(10,2)

-- 2. CREACIÓN DEL DATA WAREHOUSE (BilleteraDW) DESDE CERO
CREATE DATABASE BilleteraDW;
USE BilleteraDW;

CREATE TABLE dim_usuario (id_usuario INT PRIMARY KEY, edad INT, sexo VARCHAR(20), rol VARCHAR(20), fecha_intervencion_rag DATE);
CREATE TABLE dim_categoria (id_categoria INT PRIMARY KEY, nombre_categoria VARCHAR(100), tipo_categoria VARCHAR(50));
CREATE TABLE dim_mediopago (id_medio_pago INT PRIMARY KEY, canal_pago VARCHAR(50), tipo_canal VARCHAR(50));
CREATE TABLE dim_tiempo (id_tiempo INT PRIMARY KEY, fecha DATE, mes INT, trimestre INT, anio INT);
CREATE TABLE dim_estadofinanciero (id_estado INT PRIMARY KEY, estado VARCHAR(20), descripcion VARCHAR(255));
CREATE TABLE dim_rangomonto (id_rango INT PRIMARY KEY, rango_gasto VARCHAR(20), monto_minimo DECIMAL(10,2), monto_maximo DECIMAL(10,2));

CREATE TABLE fact_finanzas (
    id_transaccion INT PRIMARY KEY, id_usuario INT, id_categoria INT, id_medio_pago INT, id_tiempo INT,
    id_estado INT, id_rango INT, fecha_hora DATETIME,
    monto_ingreso DECIMAL(10,2), monto_gasto DECIMAL(10,2), monto_ahorro DECIMAL(10,2), monto_deuda DECIMAL(10,2),
    ratio_endeudamiento DECIMAL(5,2), probabilidad_riesgo DECIMAL(5,4)
);

-- 3. PROCEDIMIENTO ALMACENADO (ETL)
DELIMITER //
CREATE PROCEDURE sp_etl_billetera()
BEGIN
    INSERT IGNORE INTO dim_estadofinanciero VALUES (1, 'Sin Riesgo', 'Estable'), (2, 'Riesgo', 'Vulnerabilidad');
    INSERT IGNORE INTO dim_rangomonto VALUES (1, 'Bajo', 0.00, 500.00), (2, 'Medio', 500.01, 1500.00), (3, 'Alto', 1500.01, 99999.00);

    INSERT IGNORE INTO dim_usuario SELECT * FROM BilleteraOLTP.Usuario;
    INSERT IGNORE INTO dim_categoria SELECT * FROM BilleteraOLTP.Categoria;
    INSERT IGNORE INTO dim_mediopago SELECT * FROM BilleteraOLTP.MedioPago;
    
    INSERT IGNORE INTO dim_tiempo 
    SELECT DISTINCT CAST(DATE_FORMAT(FechaHora, '%Y%m%d') AS UNSIGNED), DATE(FechaHora), MONTH(FechaHora), QUARTER(FechaHora), YEAR(FechaHora) 
    FROM BilleteraOLTP.Transaccion;

    BEGIN
        DECLARE done INT DEFAULT FALSE;
        DECLARE v_IdTrans, v_IdUsr, v_IdCat, v_IdMed, v_IdTiempo, v_IdEstado, v_IdRango INT;
        DECLARE v_FechaHora DATETIME;
        DECLARE v_Ing, v_Gasto, v_Ahorro, v_Deuda DECIMAL(10,2);
        DECLARE v_Ratio DECIMAL(5,2); DECLARE v_Prob DECIMAL(5,4);

        DECLARE cur_fact CURSOR FOR SELECT IdTransaccion, IdUsuario, IdCategoria, IdMedioPago, FechaHora, MontoIngreso, MontoGasto, MontoAhorro, MontoDeuda FROM BilleteraOLTP.Transaccion;
        DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;

        OPEN cur_fact;
        read_fact: LOOP
            FETCH cur_fact INTO v_IdTrans, v_IdUsr, v_IdCat, v_IdMed, v_FechaHora, v_Ing, v_Gasto, v_Ahorro, v_Deuda;
            IF done THEN LEAVE read_fact; END IF;

            SET v_IdTiempo = CAST(DATE_FORMAT(v_FechaHora, '%Y%m%d') AS UNSIGNED);
            IF v_Gasto <= 500 THEN SET v_IdRango = 1; ELSEIF v_Gasto <= 1500 THEN SET v_IdRango = 2; ELSE SET v_IdRango = 3; END IF;
            
            IF v_Ing > 0 THEN SET v_Ratio = ROUND(v_Deuda / v_Ing, 4); ELSE SET v_Ratio = 0.00; END IF;
            IF v_Ratio > 0.35 THEN SET v_IdEstado = 2; ELSE SET v_IdEstado = 1; END IF;
            SET v_Prob = v_Ratio;

            INSERT IGNORE INTO fact_finanzas VALUES (v_IdTrans, v_IdUsr, v_IdCat, v_IdMed, v_IdTiempo, v_IdEstado, v_IdRango, v_FechaHora, v_Ing, v_Gasto, v_Ahorro, v_Deuda, v_Ratio, v_Prob);
        END LOOP;
        CLOSE cur_fact;
    END;
END //
DELIMITER ;
